cd ~/projects/code/ncnn
mkdir build-v821
cd build-v821

然后在 Tina SDK 里确认 V821 工具链路径。通常在：
<tina-sdk>/prebuilt/linux-x86/
<tina-sdk>/toolchain/
找到类似：
riscv64-unknown-linux-gnu-gcc --version     #看是64位还是32位
或riscv64-linux-gnu-gcc --version
确认前缀后，再编译 ncnn：
cmake -DCMAKE_BUILD_TYPE=Release \
      -DNCNN_VULKAN=OFF \
      -DNCNN_THREADS=OFF \
      -DNCNN_OPENMP=OFF \
      -DNCNN_SIMPLEOCV=ON \
      -DNCNN_BUILD_EXAMPLES=ON \
      -DCMAKE_C_COMPILER=riscv64-unknown-linux-gnu-gcc \
      -DCMAKE_CXX_COMPILER=riscv64-unknown-linux-gnu-g++ \
      ..

make -j$(nproc)
